\chapter{Adapti}

\begin{itemize}
\item \textbf{Overview:} Implement an adaptive architecture that starts simple (file-based only) and automatically scales to complex mode (queue + sidecar index) when needed. Threshold-based switching ensures we only add complexity when performance requires it.
\end{itemize}


ve Core Architecture Implementation Plan

\subsection{Context}

Building on the core data structure analysis, this plan implements an \textbf{adaptive architecture} that:

\begin{itemize}
\item Starts \textbf{simple} (file-based only) for small scale (< 100 work efforts)
\item Automatically switches to \textbf{complex} (queue + sidecar index) when scale requires it (>= 100 work efforts)
\item Provides seamless transition between modes
\end{enumerate}

\subsection{Architecture Modes}

\subsubsection{Tiny Mode (< 10 work efforts, no concurrency)}

\begin{itemize}
\item \textbf{Storage}: File system only
\item \textbf{Lookups}: O(n) directory scan (~0.01-0.05s)
\item \textbf{Writes}: Direct file I/O
\item \textbf{Git}: Direct git commands
\item \textbf{Use Case}: Initial setup, tiny personal projects
\end{enumerate}

\subsubsection{Simple Mode (10-20 work efforts, no concurrency)}

\begin{itemize}
\item \textbf{Storage}: File system only
\item \textbf{Lookups}: O(n) directory scan (~0.05-0.1s)
\item \textbf{Writes}: Direct file I/O (no queue)
\item \textbf{Git}: Direct git commands (no queue)
\item \textbf{Use Case}: Small projects, sequential access only
\end{enumerate}

\subsubsection{Standard Mode (>= 20 work efforts OR concurrency detected)}

\begin{itemize}
\item \textbf{Storage}: File system + Sidecar SQLite index (if >= 20 items)
\item \textbf{Lookups}: O(1) index queries (~0.005s) OR O(n) scan if index disabled
\item \textbf{Writes}: FIFO queue (always enabled if concurrency)
\item \textbf{Git}: Queued git operations (sequential processing)
\item \textbf{Use Case}: Most real-world usage (your case: medium scale + always concurrent)
\end{enumerate}

\subsubsection{Performance Mode (>= 50 work efforts)}

\begin{itemize}
\item \textbf{Storage}: File system + Sidecar SQLite index (required)
\item \textbf{Lookups}: O(1) index queries only (~0.005s)
\item \textbf{Writes}: FIFO queue (required)
\item \textbf{Git}: Queued git operations (required)
\item \textbf{Use Case}: Medium to large scale, performance critical
\end{enumerate}

\subsection{Threshold Configuration}

\begin{verbatim}
const SCALE_THRESHOLDS = {
  // Queue: Enabled based on concurrency, not scale
  QUEUE\textit{ENABLED}ALWAYS: true,   // Always enabled (concurrent writes expected)

  // Index: Enabled based on performance requirements
  INDEX\textit{ENABLED}MIN: 20,        // Enable index at 20 items for instant lookups
  INDEX\textit{REQUIRED}MIN: 50,       // Index required at 50 items (performance critical)

  // Simple Mode: Only for tiny scale with no concurrency
  SIMPLE\textit{MODE}MAX: 20,          // Simple mode only < 20 items (rare case)
  TINY\textit{MODE}MAX: 10             // No queue/index needed < 10 items
};

\end{verbatim}



\subsubsection{Rationale}

\textbf{Queue (Concurrency Safety):}

\begin{itemize}
\item \textbf{Why always enabled}: You have "always concurrent" writes (human + AI)
\item \textbf{Not scale-based}: Queue prevents git lock contention, which happens at any scale with concurrency
\item \textbf{Decision}: Enable queue from day 1 if concurrent writes detected, or always if configured
\end{enumerate}

\textbf{Index (Performance):}

\begin{itemize}
\item \textbf{Why 20 items}: For "instant" performance (< 0.01s), directory scans become too slow
\item 20 items: ~0.05-0.1s scan (not instant)
\item 50 items: ~0.1-0.2s scan (not instant)
\item Index lookup: ~0.005s (instant)
\item \textbf{Why 50 required}: At 50+ items, scans are consistently > 0.1s, index becomes essential
\item \textbf{Decision}: Enable index at 20 items, require at 50 items
\end{enumerate}

\textbf{Simple Mode:}

\begin{itemize}
\item \textbf{Why < 20 items}: Only makes sense if:
\item No concurrency (but you have "always concurrent")
\item Small scale (< 20 items)
\item Performance tolerance is higher (but you want "instant")
\item \textbf{Decision}: Simple mode is rare, mostly for initial setup or tiny projects
\end{enumerate}

\begin{enumerate}
\item Enable features:
\end{enumerate}

\begin{itemize}
\item \textbf{Queue}: Always if concurrency detected, or if count >= 20

\begin{enumerate}
\item Store mode in config file (\texttt{.pyrite/config.json})

\textbf{On Work Effort Creation:}

\begin{enumerate}
\item Check current count and concurrency status
\item If transitioning to new mode:
\end{enumerate}

\begin{itemize}
\item \textbf{10 → 20 items}: Enable index (if not already)

\begin{verbatim}
async create(data) {
  const count = await this.countWorkEfforts();
  const hasConcurrency = await this.detectConcurrency();

  // Check if we need to migrate
  if (hasConcurrency && !this.queue.isEnabled()) {
    await this.enableQueue();
  }

  if (count >= 20 && !this.index.isEnabled()) {
    await this.enableIndex();
  }

  if (count >= 50 && !this.index.isRequired()) {
    await this.requireIndex();
  }

  // Execute in current mode
  if (this.queue.isEnabled()) {
    return await this.queue.add('CREATE\textit{WORK}EFFORT', data);
  } else {
    return await this.simpleMode.create(data);
  }
}

\end{verbatim}

\begin{itemize}
\item \textbf{Binary}: Yes/No, Option A/Option B (e.g., binder vs single PDF)
\item \textbf{Multiple Choice}: Option A/B/C/D (e.g., output style)
\item \textbf{Ranked}: Rank options 1-3 (e.g., top integration opportunities)
\item \textbf{Weighted}: Assign weights to options (optional, for skill-based weighting)
\end{enumerate}

\textbf{Voting Process}:

\begin{enumerate}
\item \textbf{Command randomly selects some Beings} (mostly at random) to participate in voting
\item Selection is weighted slightly by relevance (e.g., Being with integration skills more likely selected for integration decisions)
\item Selection size: Typically 50-70% of town (e.g., 2-3 Beings from 5, or 3-4 from 7)
\item Present decision and options to selected Beings only
\item Selected Beings vote (with reasoning)
\item Other Beings are not asked (they don't participate in this vote)
\item Collect votes from selected Beings
\item Calculate results:
\end{enumerate}

                                                - Count votes (from selected Beings only)
                                                - Majority vote wins
                                                - If tie: Oracle breaks tie

\begin{enumerate}
\item Document all votes and selection process
\end{enumerate}

\textbf{Selection Documentation}:

\begin{itemize}
\item Which Beings were selected (and why - relevance weighting)
\item Which Beings were not selected (not asked to vote)
\item Selection method documented for transparency
\item All votes preserved for transparency
\end{enumerate}
