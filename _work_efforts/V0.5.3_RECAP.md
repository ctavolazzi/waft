## Complete Flow Example

```python
# 1. Source creates reality
reality_system = RealitySystem()
reality = reality_system.create_reality(
    reality_type=RealityType.LEARNING,
    configuration={"focus": "cognitive_skills"}
)
# Registered: source_consciousness → reality_123

# 2. Start reality
reality_system.start_reality(reality.reality_id)

# 3. Spawn beings into reality
being_system = BeingSystem()
being = being_system.spawn_being(
    reality_id=reality.reality_id,
    initial_skills={"reasoning": 10.0, "analysis": 5.0}
)
# Registered: source_consciousness → reality_123 → being_456

# 4. Being learns in reality
being.learn_skill("reasoning", "cognitive", 5.0)  # Now level 15.0
being.learn_skill("creativity", "creative", 3.0)   # New skill
being.learn_lesson("Approach A works better", "success")
being.record_memory("Pattern: X → Y", "pattern")

# 5. Extract memories from being's experience
memory_flow = MemoryFlow()
experience = {
    "lessons_learned": being.lessons_learned,
    "skills": being.skills,
    "patterns_discovered": ["X → Y"],
    "insights": ["Deep insight about evolution"]
}
memories = memory_flow.extract_memories_from_experience(being.being_id, experience)

# 6. Pass memories upward
result = memory_flow.pass_memories_upward(being.being_id, memories)
# Flow: being_456 → reality_123 → source_consciousness
# Capacity: lessons (2.0) + skills (3.0) + patterns (4.0) + insights (5.0) = 14.0

# 7. Complete being
being_system.complete_being(being.being_id, final_fitness=0.8)
# More capacity flows upward

# 8. End reality
reality_system.end_reality(reality.reality_id, outcomes={
    "lessons_learned": [...],
    "skills_developed": [...],
    "memories_generated": [...]
})
# Reality memories also flow upward

# 9. Source accumulates knowledge
source = SourceConsciousness()
stats = source.get_source_stats()
print(f"Source Capacity: {stats['total_capacity_accumulated']}")
print(f"Total Permutations: {stats['total_permutations']}")

# 10. Source accomplishes goal
if stats['total_capacity_accumulated'] >= 1000.0:
    result = source.accomplish_goal(
        goal_description="Understand evolution through permutation",
        required_capacity=1000.0
    )
    print("✅ Source accomplished its original goal!")
```
