% Image error handling: Use placeholder if image not found
\newcommand{\safeincludegraphics}[2][width=\textwidth]{%
	\IfFileExists{#2}{%
		\includegraphics[#1]{#2}%
	}{%
		\fbox{\parbox{0.9\textwidth}{\centering\itshape Image not found: #2}}%
		\typeout{Warning: Image '#2' not found}%
	}%
}

% Horizontal rule
\newcommand{\HorRule}[1]{\noindent\rule{\linewidth}{#1}}

% Separator rule
\newcommand{\SepRule}{\noindent
	\begin{center}
		\rule{250pt}{1pt}
	\end{center}
}

% Journal name (title)
\newcommand{\JournalName}[1]{%
	\begin{center}	
		\Huge \usefont{T1}{augie}{m}{n}
		#1%
	\end{center}	
	\par \normalsize \normalfont
}

% Journal issue (date and number)
\newcommand{\JournalIssue}[1]{%
	\hfill \textsc{\mydate \today, No #1}
	\par \normalsize \normalfont
}

% News item title
\newcommand{\NewsItem}[1]{%
	\usefont{T1}{augie}{m}{n} 	
	\large #1 \vspace{4pt}
	\par \normalsize \normalfont
}

% News author
\newcommand{\NewsAuthor}[1]{%
	\hfill by \textsc{#1} \vspace{4pt}
	\par \normalfont
}
